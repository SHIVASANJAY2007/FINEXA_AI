**MIRA Algorithm Requirements:**
MIRA maintains per-class weight vectors (5 readiness states × 10 features total: 6 recipient + 4 giver). For each family update:
1. **Predict:** Compute scores for each readiness state using current weights.
2. **Update:** If the predicted state differs from the "true" state (determined by whether the handoff would succeed given current gaps), apply the minimal weight adjustment to satisfy the margin constraint.
3. **Margin Tracking:** After each update, compute `margin = score(true_state) - max(score(other_states))`. This is the "Legacy Margin."
4. **Per-Recipient Tracking:** Maintain separate margin values for each recipient so the dashboard can show which family member needs the most attention.

# mira_handoff_planner.py
import numpy as np
from typing import Dict, List

class MIRAHandoffPlanner:
    def __init__(self, n_features=10, n_classes=5, learning_rate=0.1):
        self.w = np.zeros((n_classes, n_features))
        self.lr = learning_rate
        self.recipient_margins: Dict[str, float] = {}
        self.feature_names = [
            "recipient_age", "financial_literacy", "income_stability",
            "existing_assets", "time_horizon", "legal_structure",
            "giver_age", "estate_value", "liquidity_ratio", "will_completeness"
        ]
    
    def predict_readiness(self, x: np.ndarray) -> int:
        scores = self.w @ x
        return int(np.argmax(scores))
    
    def update(self, x: np.ndarray, true_state: int):
        """MIRA update with margin constraint."""
        scores = self.w @ x
        predicted = int(np.argmax(scores))
        if predicted == true_state:
            return
        
        margin = scores[true_state] - scores[predicted]
        if margin >= 1.0:
            return
        
        tau = (1.0 - margin) / (2.0 * np.dot(x, x) + 1e-8)
        self.w[true_state] += self.lr * tau * x
        self.w[predicted] -= self.lr * tau * x
    
    def get_legacy_margin(self, x: np.ndarray, true_state: int) -> float:
        scores = self.w @ x
        others = [scores[i] for i in range(len(scores)) if i != true_state]
        return float(scores[true_state] - max(others))
    
    def simulate_action(self, x: np.ndarray, feature_index: int, delta: float, true_state: int) -> float:
        """Simulate margin change from a single feature adjustment."""
        x_new = x.copy()
        x_new[feature_index] += delta
        return self.get_legacy_margin(x_new, true_state)
    
    def rank_actions(self, x: np.ndarray, true_state: int, actions: List[dict]) -> List[dict]:
        """Rank actions by their impact on Legacy Margin."""
        baseline = self.get_legacy_margin(x, true_state)
        ranked = []
        for action in actions:
            new_margin = self.simulate_action(x, action["feature_index"], action["delta"], true_state)
            ranked.append({**action, "margin_impact": round(new_margin - baseline, 3)})
        return sorted(ranked, key=lambda a: -a["margin_impact"])
